            function loadMap(routeData) {
                var map = new google.maps.Map(document.getElementById('map'), {
                    center: { lat: routeData.MapConfig.Center.Latitude, lng: routeData.MapConfig.Center.Longitude },
                    zoom: routeData.MapConfig.Zoom
                });

                var routePath = new google.maps.Polyline({
                    path: routeData.Coordinates.map(coord => ({ lat: coord.Latitude, lng: coord.Longitude })),
                    geodesic: true,
                    strokeColor: '#FF0000',
                    strokeOpacity: 1.0,
                    strokeWeight: 2
                });

                routePath.setMap(map);
            }

        function showMap(longitude, latitude, productId) {
            var mapOptions = {
                center: new google.maps.LatLng(latitude, longitude),
                zoom: 11
            };
            var map = new google.maps.Map(document.getElementById("mapContainer"), mapOptions);

            // Add a red pin at the specified location
            var marker = new google.maps.Marker({
                position: mapOptions.center,
                map: map,
                icon: "http://maps.google.com/mapfiles/ms/icons/red-dot.png"
            });

            // Add an info window with an image to the map
            var infoWindow = new google.maps.InfoWindow({
                content: '<img src="/EAQTA/DrawAndSend/ShowImage/' + productId + '" style="width: 300px; height: auto;"/>'
            });

            infoWindow.open(map, marker);
        }
    </script>
